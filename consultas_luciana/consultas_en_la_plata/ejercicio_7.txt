# Lo que quiero llegar a tener ==>
# ?barrio | ?precioAEstrenarPromedio | ?precioUsadoPromedio
# Centro  | 300,000                  | 250,000
# None    | 0                        | 78,900

SELECT ?departamentos ?fechaMasNueva ?valorEnDolares ?barrioLaPlata
WHERE{
  #Obtengo una tabla con cada departamento y su fecha más actual
  {
    SELECT ?departamentosFirst (MAX(?fecha) AS ?fechaMasNueva) ?valorEnDolares
    WHERE{
      ?departamentos a :Apartment .
      ?departamentos ^sioc:about ?listings .
      
      ?listings gr:hasBusinessFunction gr:Sell .
      ?listings :hasFeature ?featBlankNodePrecio .
      
      ?featBlankNodePrecio a :Price .
      ?featBlankNodePrecio time:hasTime/time:inXSDDateTimeStamp ?fecha .
      
      ?featBlankNodePrecio :hasValue ?blankNodePriceSpecification .
      ?blankNodePriceSpecification a gr:UnitPriceSpecification .
      ?blankNodePriceSpecification gr:priceType "BASE"^^xsd:string .
      ?blankNodePriceSpecification gr:hasCurrency ?monedaSinEstandar .
      ?blankNodePriceSpecification gr:hasCurrencyValue ?valorSinEstandar .

      BIND(
        IF(?monedaSinEstandar = "ARS", ?valorSinEstandar / 1375, ?valorSinEstandar)
        AS ?valorEnDolares
      )

      BIND(?departamentos as ?departamentosFirst)
    }
    GROUP BY ?departamentosFirst ?valorEnDolares
  }

  {
    SELECT DISTINCT ?departamentosSecond ?barrioLaPlata
    WHERE {
      ?departamentos a :Apartment .
      #?departamentos :hasFeature/:hasValue/:city/rdfs:label ?labelLaPlata .
      ?departamentos :hasFeature/:hasValue/:neighborhood/rdfs:label ?barrioLaPlata .
      #FILTER(CONTAINS(LCASE(STR(?labelLaPlata)), "la plata")) .
      BIND(?departamentos AS ?departamentosSecond)
      
    }
    ORDER BY ?departamentosSecond
  }

  
  FILTER(SAMETERM(?departamentosFirst, ?departamentosSecond))
  BIND(?departamentosFirst as ?departamentos)
  
}
GROUP BY ?departamentos ?fechaMasNueva ?valorEnDolares ?barrioLaPlata
ORDER BY ?departamentos
