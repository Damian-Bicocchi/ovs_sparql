# 1.  ¿Cuánto vale una casa en Berisso?


# Sigo sin poder usar un objeto especifico que
# este marcado como Berisso.
# (Esta el barrio Los Talas, pero por lo que
# busque, o es un barrio privado, lo cual me lleva el promedio
# demasiado alto, o es una localidad de 500 habitantes)

# Resuelvo con el label hardcodeado y con el label
# del listing.
SELECT (AVG(?valueNormalized) AS ?averagePriceOfBerissoHouse)
WHERE 
{
SELECT ?casa ?valueNormalized
WHERE { 
  ?casa a :House .
  ?casa :hasFeature ?featureAddress .
  ?featureAddress a :Address .
  ?featureAddress :hasValue/:address ?labelAddress .
  ?casa ^sioc:about ?listings .
  # Aca una subconsulta para obtener los listings de hace máximo 3 meses podría ir bien
  ?listings rdfs:label ?labelListing
  FILTER(
    CONTAINS(LCASE(STR(?labelAddress)), "berisso")
    ||
    CONTAINS(LCASE(STR(?labelListing)), "berisso")
  ) .
  ?listings gr:hasBusinessFunction gr:Sell .
  ?listings :hasFeature ?featurePrice .
  ?featurePrice a :Price .
  ?featurePrice :hasValue/gr:priceType "BASE"^^xsd:string .
  ?featurePrice :hasValue/gr:hasCurrency ?currencyNotNormalized .
  ?featurePrice :hasValue/gr:hasCurrencyValue ?valueNotNormalized .
  ?featurePrice time:hasTime/time:inXSDDateTimeStamp ?listingsDate
  BIND(
      IF(?currencyNotNormalized != "USD", ?valueNotNormalized / 1375, ?valueNotNormalized)
      AS ?valueNormalized
  )
  

}
GROUP BY ?casa ?valueNormalized
}
