# 4¿Qué diferencia de precio por m2 existe entre los terrenos de Villa Elvira (La Plata) y City Bell (La Plata)?
SELECT *
WHERE {
  # En Villa Elvira
  {
    SELECT (AVG(?average) AS ?averageInVillaElvira)
    WHERE {
      ?terreno a :Land .
      ?terreno :hasFeature/:hasValue/:neighborhood/rdfs:label ?labelVillaElvira .
      FILTER(CONTAINS(LCASE(STR(?labelVillaElvira)), "villa elvira")) .
      
      ?terreno ^sioc:about ?listing .

      # Obtengo el blank node del precio
      ?listing :hasFeature/:hasValue ?blankNodePrecio .
      ?blankNodePrecio a gr:UnitPriceSpecification .
      ?blankNodePrecio gr:hasCurrency ?currencyNotNormalized .
      ?blankNodePrecio gr:hasCurrencyValue ?priceNotNormalized .
      ?blankNodePrecio gr:priceType "BASE"^^xsd:string .

      BIND(
        IF(?currencyNotNormalized != "USD", ?priceNotNormalized / 1365, ?priceNotNormalized)
        AS ?priceNormalized
      )

      # Obtengo el blank node de la superficie
      ?terreno rec:includes/:hasFeature/:hasValue ?blankNodeSuperficie .
      ?blankNodeSuperficie gr:hasUnitOfMeasurement ?unitOfMeasurement .
      ?blankNodeSuperficie gr:hasValue ?measurementNotNormalized .
      FILTER(DATATYPE(?measurementNotNormalized) != xsd:string) .

      BIND (
        IF(?unitOfMeasurement = "ha", ?measurementNotNormalized * 10000, ?measurementNotNormalized )
        AS ?measurementNormalized
      )

      BIND(
        (?priceNormalized / ?measurementNormalized) AS ?average
      )

    }
  }

  # En CITY BELL
  {
    SELECT (AVG(?average) AS ?averageInCityBell)
    WHERE {
      ?terreno a :Land .
      ?terreno :hasFeature/:hasValue/:neighborhood/rdfs:label ?labelCityBell .
      FILTER(CONTAINS(LCASE(STR(?labelCityBell)), "city bell")) .
      
      ?terreno ^sioc:about ?listing .

      # Obtengo el blank node del precio
      ?listing :hasFeature/:hasValue ?blankNodePrecio .
      ?blankNodePrecio a gr:UnitPriceSpecification .
      ?blankNodePrecio gr:hasCurrency ?currencyNotNormalized .
      ?blankNodePrecio gr:hasCurrencyValue ?priceNotNormalized .
      ?blankNodePrecio gr:priceType "BASE"^^xsd:string .

      BIND(
        IF(?currencyNotNormalized != "USD", ?priceNotNormalized / 1365, ?priceNotNormalized)
        AS ?priceNormalized
      )

      # Obtengo el blank node de la superficie
      ?terreno rec:includes/:hasFeature/:hasValue ?blankNodeSuperficie .
      ?blankNodeSuperficie gr:hasUnitOfMeasurement ?unitOfMeasurement .
      ?blankNodeSuperficie gr:hasValue ?measurementNotNormalized .
      FILTER(DATATYPE(?measurementNotNormalized) != xsd:string) .

      BIND (
        IF(?unitOfMeasurement = "ha", ?measurementNotNormalized * 10000, ?measurementNotNormalized )
        AS ?measurementNormalized
      )

      BIND(
        (?priceNormalized / ?measurementNormalized) AS ?average
      )

    }
  }
}

# Resultado
