### A) ¿Qué cantidad de inmuebles se ofertan a la venta en este lugar? 54 de departamentos, 35 casas, 25 ph, 12 terrenos...
SELECT ?label (COUNT(?tipo_inmueble) AS ?cantidad)
WHERE {
  ?listings a pronto:RealEstateListing .
  ?listings sioc:about ?inmuebles .
  ?listings gr:hasBusinessFunction gr:Sell .
  ?inmuebles a ?tipo_inmueble .
  ?tipo_inmueble a owl:Class .
  ?tipo_inmueble rdfs:comment ?label .
  ?inmuebles :hasFeature/:hasValue/:neighborhood  :neighborhood_City_Bell_province_Bs.As._G.B.A._Sur_district_Bs.As._G.B.A._Sur_La_Plata .
   .
  
}
GROUP BY ?label
ORDER BY DESC(?cantidad)
