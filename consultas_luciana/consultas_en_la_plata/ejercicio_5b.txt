SELECT ?label ?numDormitorios ?numBaños (COUNT(*) AS ?cantidad)
WHERE {
  {
    SELECT ?label ?inmuebles 
           (COUNT(DISTINCT ?rooms) AS ?numDormitorios)
           (COUNT(DISTINCT ?bathrooms) AS ?numBaños)
    WHERE {
      ?listings a pronto:RealEstateListing .
      ?listings sioc:about ?inmuebles .
      ?listings gr:hasBusinessFunction gr:Sell .
      
      ?inmuebles a ?tipo_inmueble .
      ?tipo_inmueble a owl:Class .
      ?tipo_inmueble rdfs:comment ?label .
      
      ?inmuebles :hasFeature/:hasValue/:neighborhood :neighborhood_City_Bell_province_Bs.As._G.B.A._Sur_district_Bs.As._G.B.A._Sur_La_Plata .
      
      ?inmuebles rec:includes ?building .
      ?building a rec:Building .
      
      ?building brick:hasPart ?rooms .
      ?rooms a rec:Bedroom . 
      ?building brick:hasPart ?bathrooms .
      ?bathrooms a rec:Bathroom . 
      
    }
    # Lo debo agrupar por inmueble ( o sea, la casa en si. SI no, me suma TODOS los dormitorios de todas las casas en "CASA")
    GROUP BY ?label ?inmuebles
  }
}
GROUP BY ?label ?numDormitorios ?numBaños
ORDER BY DESC(?cantidad)
